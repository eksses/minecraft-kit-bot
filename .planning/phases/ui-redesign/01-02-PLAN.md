**Create Sidebar.jsx** for desktop navigation:
```jsx
import { NavLink } from 'react-router-dom';
import { LayoutDashboard, Bot, Server, Layers, MessageSquare, Settings, Package } from 'lucide-react';

const navItems = [
  { path: '/fleet', label: 'Dashboard', icon: LayoutDashboard },
  { path: '/fleet/bots', label: 'Bots', icon: Bot },
  { path: '/fleet/servers', label: 'Servers', icon: Server },
  { path: '/fleet/swarms', label: 'Swarms', icon: Layers },
  { path: '/fleet/tasks', label: 'Tasks', icon: Package },
  { path: '/chests', label: 'Chests', icon: Box },
  { path: '/kits', label: 'Order Kit', icon: ShoppingBag },
  { path: '/chat', label: 'Chat', icon: MessageSquare },
  { path: '/settings', label: 'Settings', icon: Settings },
];

function Sidebar({ currentPath }) {
  return (
    <nav className="sidebar">
      <div className="sidebar-header">
        <span className="text-heading">MDB</span>
      </div>
      <ul className="sidebar-nav">
        {navItems.map(({ path, label, icon: Icon }) => (
          <li key={path}>
            <NavLink
              to={path}
              className={`nav-item ${currentPath === path ? 'active' : ''}`}
            >
              <Icon size={20} />
              <span>{label}</span>
            </NavLink>
          </li>
        ))}
      </ul>
    </nav>
  );
}

export default Sidebar;
```

}

export default BottomNav;
