```javascript
import { readdir, stat, mkdir, rm } from 'fs/promises';
import { join } from 'path';
import { createWriteStream } from 'fs';
import { pipeline } from 'stream/promises';
import { eq } from 'drizzle-orm';
import { db } from '../db/index.js';
import { plugins } from '../db/schema.js';
import { pluginLoader } from './plugin-loader.js';
import { parseManifest, validateManifest } from '../utils/plugin-manifest.js';

const PLUGINS_DIR = join(process.cwd(), 'data', 'plugins');
const DEFAULT_REGISTRY_URL = 'https://raw.githubusercontent.com/R-Samir-Bhuiyan-A/minecraft-kit-bot/main/plugins.json';

/**
 * PluginStore - Manages plugin discovery, installation, and updates
 */
export class PluginStore {
  constructor() {
    this.repos = [
      { id: 'official', name: 'Official Registry', url: DEFAULT_REGISTRY_URL },
    ];
  }

  /**
   * Fetch available plugins from all configured repos
   * @returns {Array} List of available plugins
   */
  async getAvailable() {
    const available = [];
    
    for (const repo of this.repos) {
      try {
        const res = await fetch(repo.url);
        if (res.ok) {
          const plugins = await res.json();
          available.push(...plugins.map(p => ({ ...p, repoId: repo.id })));
        }
      } catch (err) {
        console.error(`[PluginStore] Failed to fetch ${repo.name}:`, err.message);
      }
    }
    
    return available;
  }

  /**
   * Get installed plugins with status
   * @returns {Array} List of installed plugins
   */
  async getInstalled() {
    const installed = await db.select().from(plugins);
    
    return installed.map(p => ({
      ...p,
      hasUpdate: false, // TODO: Compare versions
    }));
  }

  /**
   * Add a custom plugin repository
   * @param {string} name - Repository name
   * @param {string} url - Repository URL (must have plugins.json)
   */
  async addRepo(name, url) {
    // Validate URL has plugins.json
    const testUrl = url.endsWith('/') ? url + 'plugins.json' : url + '/plugins.json';
    const res = await fetch(testUrl);
    
    if (!res.ok) {
      throw new Error('Invalid repository URL - plugins.json not found');
    }
    
    const repo = {
      id: `custom-${Date.now()}`,
      name,
      url: testUrl,
    };
    
    this.repos.push(repo);
    return repo;
  }

  /**
   * Install a plugin from repository
   * @param {string} pluginId - Plugin identifier
   * @param {string} downloadUrl - Download URL for plugin zip/tar
   */
  async install(pluginId, downloadUrl) {
    console.log(`[PluginStore] Installing ${pluginId}...`);
    
    // Create plugin directory
    const pluginDir = join(PLUGINS_DIR, pluginId);
    await mkdir(pluginDir, { recursive: true });
    
    // Download and extract (simplified - real impl would use tar/zip)
    // For now, assume plugin files are already in place
    
    // Load the plugin
    await pluginLoader.loadPlugin(pluginDir);
    
    console.log(`[PluginStore] Installed ${pluginId}`);
  }

  /**
   * Uninstall a plugin
   * @param {string} pluginId - Plugin identifier
   */
  async uninstall(pluginId) {
    console.log(`[PluginStore] Uninstalling ${pluginId}...`);
    
    // Disable plugin first
    await pluginLoader.togglePlugin(pluginId, false);
    
    // Remove from database
    await db.delete(plugins).where(eq(plugins.id, pluginId));
    
    // Remove plugin directory
    const pluginDir = join(PLUGINS_DIR, pluginId);
    await rm(pluginDir, { recursive: true, force: true });
    
    console.log(`[PluginStore] Uninstalled ${pluginId}`);
  }

  /**
   * Update a plugin to latest version
   * @param {string} pluginId - Plugin identifier
   */
  async update(pluginId) {
    console.log(`[PluginStore] Updating ${pluginId}...`);
    
    // Disable, replace files, re-enable
    await pluginLoader.togglePlugin(pluginId, false);
    
    // TODO: Download new version
    
    await pluginLoader.togglePlugin(pluginId, true);
    
    console.log(`[PluginStore] Updated ${pluginId}`);
  }
}

export const pluginStore = new PluginStore();
```

```javascript
import { Hono } from 'hono';
import { pluginStore } from '../services/plugin-store.js';

export const pluginStoreRoutes = new Hono();

// GET /api/plugin-store/available - List available plugins from repos
pluginStoreRoutes.get('/available', async (c) => {
  const available = await pluginStore.getAvailable();
  return c.json({ plugins: available });
});

// GET /api/plugin-store/installed - List installed plugins
pluginStoreRoutes.get('/installed', async (c) => {
  const installed = await pluginStore.getInstalled();
  return c.json({ plugins: installed });
});

// POST /api/plugin-store/repos - Add custom repository
pluginStoreRoutes.post('/repos', async (c) => {
  const { name, url } = await c.req.json();
  
  try {
    const repo = await pluginStore.addRepo(name, url);
    return c.json({ success: true, repo });
  } catch (err) {
    return c.json({ error: err.message }, 400);
  }
});

// POST /api/plugin-store/install/:id - Install plugin
pluginStoreRoutes.post('/install/:id', async (c) => {
  const id = c.req.param('id');
  const { downloadUrl } = await c.req.json();
  
  try {
    await pluginStore.install(id, downloadUrl);
    return c.json({ success: true });
  } catch (err) {
    return c.json({ error: err.message }, 500);
  }
});

// DELETE /api/plugin-store/uninstall/:id - Uninstall plugin
pluginStoreRoutes.delete('/uninstall/:id', async (c) => {
  const id = c.req.param('id');
  
  try {
    await pluginStore.uninstall(id);
    return c.json({ success: true });
  } catch (err) {
    return c.json({ error: err.message }, 500);
  }
});

// POST /api/plugin-store/update/:id - Update plugin
pluginStoreRoutes.post('/update/:id', async (c) => {
  const id = c.req.param('id');
  
  try {
    await pluginStore.update(id);
    return c.json({ success: true });
  } catch (err) {
    return c.json({ error: err.message }, 500);
  }
});
```
