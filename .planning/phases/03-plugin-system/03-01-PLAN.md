import { readFileSync } from 'fs';
import { join } from 'path';

/**
 * Parse plugin.json manifest from plugin directory
 * @param {string} pluginDir - Path to plugin directory
 * @returns {object} Parsed manifest
 */
export function parseManifest(pluginDir) {
  const manifestPath = join(pluginDir, 'plugin.json');
  const raw = readFileSync(manifestPath, 'utf-8');
  return JSON.parse(raw);
}

/**
 * Validate plugin manifest has required fields
 * @param {object} manifest - Parsed manifest
 * @returns {{ valid: boolean, errors: string[] }}
 */
export function validateManifest(manifest) {
  const errors = [];
  const required = ['id', 'name', 'version', 'entry'];
  
  for (const field of required) {
    if (!manifest[field]) {
      errors.push(`Missing required field: ${field}`);
    }
  }
  
  if (manifest.version && !/^\d+\.\d+\.\d+$/.test(manifest.version)) {
    errors.push('Version must be semver format (x.y.z)');
  }
  
  if (manifest.dependencies && typeof manifest.dependencies !== 'object') {
    errors.push('Dependencies must be an object');
  }
  
  return { valid: errors.length === 0, errors };
}
```

import { readdir, stat } from 'fs/promises';
import { join } from 'path';
import { Worker } from 'worker_threads';
import { eq } from 'drizzle-orm';
import { db } from '../db/index.js';
import { plugins, pluginSettings } from '../db/schema.js';
import { parseManifest, validateManifest } from '../utils/plugin-manifest.js';

const PLUGINS_DIR = join(process.cwd(), 'data', 'plugins');

export class PluginLoader {
  constructor() {
    this.pluginWorkers = new Map(); // pluginId -> Worker
    this.pluginContexts = new Map(); // pluginId -> context
  }

  /**
   * Scan and load all enabled plugins on startup
   */
  async loadAll() {
    console.log('[PluginLoader] Scanning plugins directory...');
    
    const entries = await readdir(PLUGINS_DIR, { withFileTypes: true });
    const pluginDirs = entries.filter(e => e.isDirectory());
    
    for (const dir of pluginDirs) {
      const pluginDir = join(PLUGINS_DIR, dir.name);
      try {
        await this.loadPlugin(pluginDir);
      } catch (err) {
        console.error(`[PluginLoader] Failed to load ${dir.name}:`, err.message);
      }
    }
    
    console.log(`[PluginLoader] Loaded ${this.pluginWorkers.size} plugins`);
  }

  /**
   * Load a single plugin from directory
   */
  async loadPlugin(pluginDir) {
    const manifest = parseManifest(pluginDir);
    const { valid, errors } = validateManifest(manifest);
    
    if (!valid) {
      throw new Error(`Invalid manifest: ${errors.join(', ')}`);
    }
    
    // Check if plugin is enabled in database
    const [existing] = await db.select()
      .from(plugins)
      .where(eq(plugins.id, manifest.id))
      .limit(1);
    
    if (existing && !existing.enabled) {
      console.log(`[PluginLoader] Plugin ${manifest.id} is disabled, skipping`);
      return;
    }
    
    // Upsert plugin record
    await db.insert(plugins).values({
      id: manifest.id,
      name: manifest.name,
      version: manifest.version,
      description: manifest.description,
      author: manifest.author,
      enabled: true,
      installedAt: new Date(),
      updatedAt: new Date(),
    }).onConflictDoUpdate({
      target: plugins.id,
      set: { version: manifest.version, updatedAt: new Date() },
    });
    
    // Spawn worker thread for plugin
    await this.spawnWorker(manifest.id, pluginDir, manifest);
  }

  /**
   * Spawn a worker thread for a plugin
   */
  async spawnWorker(pluginId, pluginDir, manifest) {
    const entryPath = join(pluginDir, manifest.entry);
    
    const worker = new Worker(entryPath, {
      workerData: {
        pluginId,
        pluginDir,
        manifest,
      },
    });
    
    worker.on('error', (err) => {
      console.error(`[PluginLoader] Worker ${pluginId} error:`, err);
    });
    
    worker.on('exit', (code) => {
      console.log(`[PluginLoader] Worker ${pluginId} exited with code ${code}`);
      this.pluginWorkers.delete(pluginId);
    });
    
    this.pluginWorkers.set(pluginId, worker);
    console.log(`[PluginLoader] Spawned worker for ${pluginId}`);
  }

  /**
   * Enable/disable a plugin (hot reload)
   */
  async togglePlugin(pluginId, enabled) {
    await db.update(plugins)
      .set({ enabled, updatedAt: new Date() })
      .where(eq(plugins.id, pluginId));
    
    if (enabled) {
      // Re-load plugin
      const pluginDir = join(PLUGINS_DIR, pluginId);
      await this.loadPlugin(pluginDir);
    } else {
      // Terminate worker
      const worker = this.pluginWorkers.get(pluginId);
      if (worker) {
        worker.terminate();
        this.pluginWorkers.delete(pluginId);
      }
    }
  }

  /**
   * Get all loaded plugins
   */
  getLoadedPlugins() {
    return Array.from(this.pluginWorkers.keys());
  }
}

export const pluginLoader = new PluginLoader();
```
