/**
 * PluginUIRegistry - In-memory registry for plugin UI elements
 * Plugins register UI elements, frontend fetches and renders them
 */
export class PluginUIRegistry {
  constructor() {
    this.navItems = new Map(); // pluginId -> navItem[]
    this.routes = new Map();   // pluginId -> route[]
    this.settingsPanels = new Map(); // pluginId -> settingsPanel[]
    this.dashboardWidgets = new Map(); // pluginId -> widget[]
  }

  /**
   * Register navigation item from plugin
   * @param {string} pluginId - Plugin identifier
   * @param {object} item - { id, label, icon, path, position }
   */
  addNavItem(pluginId, item) {
    if (!this.navItems.has(pluginId)) {
      this.navItems.set(pluginId, []);
    }
    this.navItems.get(pluginId).push({
      ...item,
      pluginId,
      type: 'navItem',
    });
    console.log(`[PluginUI] Registered nav item: ${item.id} from ${pluginId}`);
  }

  /**
   * Register route from plugin
   * @param {string} pluginId - Plugin identifier
   * @param {object} route - { path, component, layout }
   */
  addRoute(pluginId, route) {
    if (!this.routes.has(pluginId)) {
      this.routes.set(pluginId, []);
    }
    this.routes.get(pluginId).push({
      ...route,
      pluginId,
      type: 'route',
    });
    console.log(`[PluginUI] Registered route: ${route.path} from ${pluginId}`);
  }

  /**
   * Register settings panel from plugin
   * @param {string} pluginId - Plugin identifier
   * @param {object} panel - { id, title, component }
   */
  addSettingsPanel(pluginId, panel) {
    if (!this.settingsPanels.has(pluginId)) {
      this.settingsPanels.set(pluginId, []);
    }
    this.settingsPanels.get(pluginId).push({
      ...panel,
      pluginId,
      type: 'settingsPanel',
    });
    console.log(`[PluginUI] Registered settings panel: ${panel.id} from ${pluginId}`);
  }

  /**
   * Register dashboard widget from plugin
   * @param {string} pluginId - Plugin identifier
   * @param {object} widget - { id, title, component, size }
   */
  addDashboardWidget(pluginId, widget) {
    if (!this.dashboardWidgets.has(pluginId)) {
      this.dashboardWidgets.set(pluginId, []);
    }
    this.dashboardWidgets.get(pluginId).push({
      ...widget,
      pluginId,
      type: 'dashboardWidget',
    });
    console.log(`[PluginUI] Registered widget: ${widget.id} from ${pluginId}`);
  }

  /**
   * Get all registered UI elements for frontend
   * @returns {object} Registry snapshot
   */
  getRegistry() {
    return {
      navItems: Array.from(this.navItems.values()).flat(),
      routes: Array.from(this.routes.values()).flat(),
      settingsPanels: Array.from(this.settingsPanels.values()).flat(),
      dashboardWidgets: Array.from(this.dashboardWidgets.values()).flat(),
    };
  }

  /**
   * Clear all registrations for a plugin (on uninstall/disable)
   * @param {string} pluginId - Plugin identifier
   */
  clearPlugin(pluginId) {
    this.navItems.delete(pluginId);
    this.routes.delete(pluginId);
    this.settingsPanels.delete(pluginId);
    this.dashboardWidgets.delete(pluginId);
    console.log(`[PluginUI] Cleared registrations for ${pluginId}`);
  }
}

export const pluginUIRegistry = new PluginUIRegistry();
```

import { useState, useEffect } from 'react';
import { Routes, Route } from 'react-router-dom';

/**
 * PluginLoader - Fetches and renders plugin UI elements
 * Called from App.jsx to inject plugin routes and nav items
 */
export function PluginLoader() {
  const [registry, setRegistry] = useState({
    navItems: [],
    routes: [],
    settingsPanels: [],
    dashboardWidgets: [],
  });

  useEffect(() => {
    fetchRegistry();
  }, []);

  const fetchRegistry = async () => {
    try {
      const res = await fetch('/api/plugin-ui/registry');
      if (res.ok) {
        const data = await res.json();
        setRegistry(data);
      }
    } catch (err) {
      console.error('[PluginLoader] Failed to fetch registry:', err);
    }
  };

  // Render plugin routes
  return (
    <Routes>
      {registry.routes.map((route) => (
        <Route
          key={route.path}
          path={route.path}
          element={<PluginRouteWrapper route={route} />}
        />
      ))}
    </Routes>
  );
}

/**
 * Wrapper for plugin routes - loads plugin component dynamically
 */
function PluginRouteWrapper({ route }) {
  const [Component, setComponent] = useState(null);

  useEffect(() => {
    // Dynamic import of plugin component
    // Component path comes from plugin registration
    import(`../plugins/${route.pluginId}/${route.component}`)
      .then((mod) => setComponent(() => mod.default))
      .catch((err) => {
        console.error(`[PluginLoader] Failed to load ${route.component}:`, err);
      });
  }, [route.pluginId, route.component]);

  if (!Component) {
    return <div>Loading plugin...</div>;
  }

  return <Component />;
}

/**
 * Hook to get plugin nav items for navigation components
 */
export function usePluginNavItems() {
  const [navItems, setNavItems] = useState([]);

  useEffect(() => {
    fetch('/api/plugin-ui/nav-items')
      .then((res) => res.json())
      .then((data) => setNavItems(data.navItems || []))
      .catch((err) => console.error('[PluginLoader] Failed to fetch nav items:', err));
  }, []);

  return navItems;
}

/**
 * Hook to get plugin dashboard widgets
 */
export function usePluginWidgets() {
  const [widgets, setWidgets] = useState([]);

  useEffect(() => {
    fetch('/api/plugin-ui/widgets')
      .then((res) => res.json())
      .then((data) => setWidgets(data.dashboardWidgets || []))
      .catch((err) => console.error('[PluginLoader] Failed to fetch widgets:', err));
  }, []);

  return widgets;
}
```
