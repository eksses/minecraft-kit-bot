```javascript
    },
  };
}

/**
 * PluginAPI service for managing plugin contexts
 */
export class PluginAPI {
  constructor() {
    this.contexts = new Map();
  }
  
  /**
   * Initialize context for a loaded plugin
   */
  initContext(pluginId, pluginDir) {
    const ctx = createPluginContext(pluginId, pluginDir);
    this.contexts.set(pluginId, ctx);
    return ctx;
  }
  
  /**
   * Get context for a plugin
   */
  getContext(pluginId) {
    return this.contexts.get(pluginId);
  }
  
  /**
   * Mount plugin routes to main app
   */
  mountRoutes(pluginId, mainApp) {
    const ctx = this.contexts.get(pluginId);
    if (ctx && ctx.app) {
      mainApp.route(`/api/plugins/${pluginId}`, ctx.app);
    }
  }
}

export const pluginAPI = new PluginAPI();
```

```javascript
import { Hono } from 'hono';
import { eq } from 'drizzle-orm';
import { db } from '../db/index.js';
import { plugins, pluginSettings } from '../db/schema.js';
import { pluginLoader } from '../services/plugin-loader.js';
import { pluginAPI } from '../services/plugin-api.js';

export const pluginRoutes = new Hono();

// GET /api/plugins - List all installed plugins
pluginRoutes.get('/', async (c) => {
  const allPlugins = await db.select().from(plugins);
  return c.json({ plugins: allPlugins });
});

// GET /api/plugins/:id - Get plugin details
pluginRoutes.get('/:id', async (c) => {
  const id = c.req.param('id');
  const [plugin] = await db.select()
    .from(plugins)
    .where(eq(plugins.id, id))
    .limit(1);
  
  if (!plugin) {
    return c.json({ error: 'Plugin not found' }, 404);
  }
  
  return c.json({ plugin });
});

// PUT /api/plugins/:id/toggle - Enable/disable plugin (D-08 hot reload)
pluginRoutes.put('/:id/toggle', async (c) => {
  const id = c.req.param('id');
  const { enabled } = await c.req.json();
  
  await pluginLoader.togglePlugin(id, enabled);
  
  return c.json({ success: true, enabled });
});

// GET /api/plugins/:id/settings - Get plugin settings (D-07)
pluginRoutes.get('/:id/settings', async (c) => {
  const id = c.req.param('id');
  const settings = await db.select()
    .from(pluginSettings)
    .where(eq(pluginSettings.pluginId, id));
  
  // Convert to key-value object
  const settingsObj = settings.reduce((acc, s) => {
    acc[s.key] = JSON.parse(s.value);
    return acc;
  }, {});
  
  return c.json({ settings: settingsObj });
});

// PUT /api/plugins/:id/settings - Update plugin settings (D-07)
pluginRoutes.put('/:id/settings', async (c) => {
  const id = c.req.param('id');
  const updates = await c.req.json();
  
  for (const [key, value] of Object.entries(updates)) {
    await db.insert(pluginSettings).values({
      pluginId: id,
      key,
      value: JSON.stringify(value),
      createdAt: new Date(),
      updatedAt: new Date(),
    }).onConflictDoUpdate({
      target: [pluginSettings.pluginId, pluginSettings.key],
      set: { value: JSON.stringify(value), updatedAt: new Date() },
    });
  }
  
  return c.json({ success: true });
});

// POST /api/plugins/:id/install - Install plugin from manifest
```
